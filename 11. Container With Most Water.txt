class Solution {
    public int maxArea(int[] height) {

        int i =0;
        int j = height.length-1;
        int res = 0;

        while(i<=j){
            int h1 = height[i];
            int h2 = height[j];

            int hei = (int) Math.min(h1,h2);

            int length = j-i;

            int area = length * hei;
            if(res<area){
                res = area;
            }
            
            if(h1<h2){
                i++;
            }else{
                j--;
            }
        }
        return res;
    }
}
